			if (handler == null) {
				handler = factory.getHandler(null);
			}
			else {

			}
			if (handler instanceof Closeable closeable) {

			}

				Assert.isNull(this.gracefulShutdown, "Only a single GracefulShutdownHandler can be defined");
				this.gracefulShutdown = shutdownHandler;
			}
		}
		return handler;
	}
